cmake_minimum_required(VERSION 3.16)
project(semicolon LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(SEMICOLON_DESKTOP "Build the OpenCV desktop demo and tests" OFF)
option(SEMICOLON_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    add_compile_options(-Wall -Wextra -Wpedantic)
    if(SEMICOLON_SANITIZERS)
        add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined)
    endif()
endif()
enable_testing()
add_library(semicolon_core lib/semicolon_core/src/semicolon_core.c)
target_include_directories(semicolon_core PUBLIC lib/semicolon_core/src)
add_executable(core_tests tests/test_core.c)
target_link_libraries(core_tests PRIVATE semicolon_core)
add_test(NAME core COMMAND core_tests)

if(SEMICOLON_DESKTOP)
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui)
    add_library(currency_detector desktop/currency_detection/currency_detector.cpp)
    target_include_directories(currency_detector PUBLIC desktop/currency_detection)
    target_include_directories(currency_detector SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(currency_detector PUBLIC ${OpenCV_LIBS})
    add_executable(currency_demo desktop/currency_detection/main.c)
    target_link_libraries(currency_demo PRIVATE currency_detector)
    add_executable(currency_tests tests/test_currency.cpp)
    target_link_libraries(currency_tests PRIVATE currency_detector)
    add_test(NAME currency COMMAND currency_tests)
    add_test(NAME desktop_help COMMAND currency_demo --help)
    add_test(NAME desktop_missing_dataset COMMAND currency_demo does-not-exist)
    set_tests_properties(desktop_missing_dataset PROPERTIES WILL_FAIL TRUE)
    add_test(NAME desktop_invalid_camera COMMAND currency_demo . -1)
    set_tests_properties(desktop_invalid_camera PROPERTIES WILL_FAIL TRUE)
endif()
